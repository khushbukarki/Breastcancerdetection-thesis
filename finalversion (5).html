# --- helpers ---
def load_rgb(path):
    img = Image.open(path)
    if img.mode != "RGB":
        img = img.convert("RGB")
    return img

@torch.no_grad()
def extract_timm_feats(df_part, model, tfm, batch=BATCH):
    model.eval()
    paths = df_part["img_full"].tolist()
    feats = []

    for i in tqdm(range(0, len(paths), batch), desc="timm-feats"):
        batch_paths = paths[i:i+batch]
        x = torch.stack([tfm(load_rgb(p)) for p in batch_paths]).to(device)
        f = model(x).detach().cpu().numpy()
        feats.append(f)

    return np.vstack(feats)

@torch.no_grad()
def extract_clip_feats(df_part, clip_model, clip_preprocess, batch=BATCH):
    clip_model.eval()
    paths = df_part["img_full"].tolist()
    feats = []

    for i in tqdm(range(0, len(paths), batch), desc="clip-feats"):
        batch_paths = paths[i:i+batch]
        x = torch.stack([clip_preprocess(load_rgb(p)) for p in batch_paths]).to(device)
        f = clip_model.encode_image(x)
        f = f / f.norm(dim=-1, keepdim=True)
        feats.append(f.detach().cpu().numpy())

    return np.vstack(feats)

    confusion_matrix, roc_auc_score, accuracy_score,
    precision_score, recall_score, f1_score, balanced_accuracy_score

def eval_at(name, X, y, thr):
    prob = rf.predict_proba(X)[:, 1]
    pred = (prob >= thr).astype(int)

    cm = confusion_matrix(y, pred, labels=[0, 1])
    tn, fp, fn, tp = cm.ravel()

    # Core screening metrics
    sensitivity = tp / (tp + fn + 1e-12)   # TPR, Recall for malignant (class 1)
    specificity = tn / (tn + fp + 1e-12)   # TNR, Recall for benign (class 0)

    # Extra helpful medical metrics
    ppv = tp / (tp + fp + 1e-12)           # Positive Predictive Value (Precision)
    npv = tn / (tn + fn + 1e-12)           # Negative Predictive Value

    auc  = roc_auc_score(y, prob)
    acc  = accuracy_score(y, pred)
    bacc = balanced_accuracy_score(y, pred)
    f1   = f1_score(y, pred, zero_division=0)

    print(f"\n== {name} @ thr={thr:.2f} ==")
    print(f"AUC        : {auc:.4f}")
    print(f"Accuracy   : {acc:.4f}")
    print(f"Bal-Acc    : {bacc:.4f}")
    print(f"Sensitivity: {sensitivity:.4f}  (TPR, malignant recall)")
    print(f"Specificity: {specificity:.4f}  (TNR, benign recall)")
    print(f"PPV (Prec) : {ppv:.4f}")
    print(f"NPV        : {npv:.4f}")
    print(f"F1         : {f1:.4f}")
    print("Confusion Matrix [ [TN FP] [FN TP] ]:")
    print(cm)

def plot_all(name, X, y, thr):

    prob = rf.predict_proba(X)[:, 1]
    pred = (prob >= thr).astype(int)

    fpr, tpr, _ = roc_curve(y, prob)
    roc_auc = auc(fpr, tpr)

    prec, rec, _ = precision_recall_curve(y, prob)
    ap = average_precision_score(y, prob)

    cm = confusion_matrix(y, pred, labels=[0, 1])

    # 🔥 4 plots side-by-side
    fig, axes = plt.subplots(1, 4, figsize=(22, 5))

    # --- ROC ---
    axes[0].plot(fpr, tpr)
    axes[0].plot([0, 1], [0, 1], linestyle="--")
    axes[0].set_title(f"{name} ROC\nAUC={roc_auc:.3f}")
    axes[0].set_xlabel("FPR")
    axes[0].set_ylabel("TPR")

    # --- PR ---
    axes[1].plot(rec, prec)
    axes[1].set_title(f"{name} PR\nAP={ap:.3f}")
    axes[1].set_xlabel("Recall")
    axes[1].set_ylabel("Precision")

    # --- Confusion matrix ---
    disp = ConfusionMatrixDisplay(cm, display_labels=["Benign", "Malignant"])
    disp.plot(ax=axes[2], values_format="d", colorbar=False)
    axes[2].set_title("Confusion Matrix")

    # --- Probability histogram ---
    axes[3].hist(prob[y == 0], bins=25, alpha=0.7, label="Benign")
    axes[3].hist(prob[y == 1], bins=25, alpha=0.7, label="Malignant")

    axes[3].set_title("Score Distribution")

import torch
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms
from PIL import Image

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Device:", device)

class ImgDataset(Dataset):
    def __init__(self, df, img_col, y_col, tfm):
        self.df = df.reset_index(drop=True)
        self.img_col = img_col
        self.y_col = y_col
        self.tfm = tfm

    def __len__(self):
        return len(self.df)

    def __getitem__(self, i):
        p = self.df.loc[i, self.img_col]
        y = int(self.df.loc[i, self.y_col])
        img = Image.open(p).convert("RGB")
        img = self.tfm(img)
        return img, y

tfm = transforms.Compose([
    transforms.Resize((224,224)),
    transforms.ToTensor(),
    transforms.Normalize(mean=(0.5,0.5,0.5), std=(0.5,0.5,0.5)),
])

import numpy as np
import matplotlib.pyplot as plt

from sklearn.metrics import (
    confusion_matrix, roc_curve, auc,
    precision_recall_curve, average_precision_score
)
from sklearn.calibration import calibration_curve

def plot_side_by_side_results(model, Xva, yva, Xte, yte, thr=0.32, title="RF (Hybrid Features)"):
    # probabilities
    pva = model.predict_proba(Xva)[:, 1]
    pte = model.predict_proba(Xte)[:, 1]

    # predictions at threshold
    pred_va = (pva >= thr).astype(int)
    pred_te = (pte >= thr).astype(int)

    # confusion matrices
    cm_va = confusion_matrix(yva, pred_va, labels=[0, 1])
    cm_te = confusion_matrix(yte, pred_te, labels=[0, 1])

    cm_va_norm = cm_va / (cm_va.sum(axis=1, keepdims=True) + 1e-12)
    cm_te_norm = cm_te / (cm_te.sum(axis=1, keepdims=True) + 1e-12)

    # ROC
    fpr_va, tpr_va, _ = roc_curve(yva, pva)
    fpr_te, tpr_te, _ = roc_curve(yte, pte)
    auc_va = auc(fpr_va, tpr_va)
    auc_te = auc(fpr_te, tpr_te)

    # PR
    prec_va, rec_va, _ = precision_recall_curve(yva, pva)
    prec_te, rec_te, _ = precision_recall_curve(yte, pte)
    ap_va = average_precision_score(yva, pva)
    ap_te = average_precision_score(yte, pte)

    # Calibration (optional but good)
    frac_pos_va, mean_pred_va = calibration_curve(yva, pva, n_bins=10, strategy="uniform")
    frac_pos_te, mean_pred_te = calibration_curve(yte, pte, n_bins=10, strategy="uniform")

    # -------------------- 1) Confusion Matrix (Counts) side by side --------------------
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    for ax, cm, name in [(axes[0], cm_va, "VAL"), (axes[1], cm_te, "TEST")]:
        im = ax.imshow(cm)
        ax.set_title(f"{title} | {name} CM (counts) @ thr={thr:.2f}")
        ax.set_xticks([0, 1]); ax.set_yticks([0, 1])
        ax.set_xticklabels(["Benign(0)", "Malignant(1)"])
        ax.set_yticklabels(["Benign(0)", "Malignant(1)"])
        ax.set_xlabel("Predicted"); ax.set_ylabel("True")
        for (i, j), v in np.ndenumerate(cm):
            ax.text(j, i, str(v), ha="center", va="center")
    plt.tight_layout()
    plt.show()

    # -------------------- 2) Confusion Matrix (Normalized) side by side --------------------
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    for ax, cmn, name in [(axes[0], cm_va_norm, "VAL"), (axes[1], cm_te_norm, "TEST")]:
        im = ax.imshow(cmn, vmin=0, vmax=1)
        ax.set_title(f"{title} | {name} CM (normalized) @ thr={thr:.2f}")
        ax.set_xticks([0, 1]); ax.set_yticks([0, 1])
        ax.set_xticklabels(["Benign(0)", "Malignant(1)"])
        ax.set_yticklabels(["Benign(0)", "Malignant(1)"])
        ax.set_xlabel("Predicted"); ax.set_ylabel("True")
        for (i, j), v in np.ndenumerate(cmn):
            ax.text(j, i, f"{v:.2f}", ha="center", va="center")
    plt.tight_layout()
    plt.show()

    # -------------------- 3) ROC Curves side by side --------------------
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    axes[0].plot(fpr_va, tpr_va, label=f"AUC={auc_va:.4f}")
    axes[0].plot([0, 1], [0, 1], linestyle="--")
    axes[0].set_title(f"{title} | VAL ROC")
    axes[0].set_xlabel("FPR (1 - Specificity)")
    axes[0].set_ylabel("TPR (Sensitivity)")
    axes[0].legend(loc="lower right")

    axes[1].plot(fpr_te, tpr_te, label=f"AUC={auc_te:.4f}")
    axes[1].plot([0, 1], [0, 1], linestyle="--")
    axes[1].set_title(f"{title} | TEST ROC")
    axes[1].set_xlabel("FPR (1 - Specificity)")
    axes[1].set_ylabel("TPR (Sensitivity)")
    axes[1].legend(loc="lower right")

    plt.tight_layout()
    plt.show()

    # -------------------- 4) PR Curves side by side --------------------
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    axes[0].plot(rec_va, prec_va, label=f"AP={ap_va:.4f}")
    axes[0].set_title(f"{title} | VAL Precision–Recall")
    axes[0].set_xlabel("Recall (Sensitivity)")
    axes[0].set_ylabel("Precision (PPV)")
    axes[0].legend(loc="lower left")

    axes[1].plot(rec_te, prec_te, label=f"AP={ap_te:.4f}")
    axes[1].set_title(f"{title} | TEST Precision–Recall")
    axes[1].set_xlabel("Recall (Sensitivity)")
    axes[1].set_ylabel("Precision (PPV)")
    axes[1].legend(loc="lower left")

    plt.tight_layout()
    plt.show()

    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    for ax, probs, y, name in [(axes[0], pva, yva, "VAL"), (axes[1], pte, yte, "TEST")]:
        ax.hist(probs[y == 0], bins=30, alpha=0.7, label="Benign(0)")
        ax.hist(probs[y == 1], bins=30, alpha=0.7, label="Malignant(1)")

        ax.set_xlabel("Predicted P(malignant)")
        ax.set_ylabel("Count")
        ax.legend()
    plt.tight_layout()
    plt.show()

    fig, axes = plt.subplots(1, 2, figsize=(12, 5))

    axes[1].set_title(f"{title} | TEST Calibration")
    axes[1].set_xlabel("Mean predicted probability")
    axes[1].set_ylabel("Fraction of positives")
    axes[1].legend()

    plt.tight_layout()
    plt.show()

    return pva, pte

def pretty_eval(name, model, X, y, thr):
    prob = model.predict_proba(X)[:, 1]
    pred = (prob >= thr).astype(int)

    cm = confusion_matrix(y, pred, labels=[0, 1])
    tn, fp, fn, tp = cm.ravel()

    sens = tp / (tp + fn + 1e-12)  # sensitivity = recall for class 1
    spec = tn / (tn + fp + 1e-12)  # specificity = recall for class 0

    print(f"\n== {name} @ thr={thr:.2f} ==")
    print("AUC :", roc_auc_score(y, prob))
    print("ACC :", accuracy_score(y, pred))
    print("PREC:", precision_score(y, pred, zero_division=0))
    print("SENS:", sens, "(Sensitivity)")
    print("SPEC:", spec)
    print("F1  :", f1_score(y, pred, zero_division=0))
    print("Confusion Matrix [ [TN FP] [FN TP] ]:\n", cm)
    print("\nReport:\n", classification_report(y, pred, digits=4))
